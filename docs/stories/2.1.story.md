## Acceptance Criteria

1. `POST /ingest/skills` endpoint created (protected by JWT middleware)
2. `POST /ingest/jobs` endpoint created (protected by JWT middleware)
3. Endpoints accept `multipart/form-data` with file field named `file`
4. File size limit enforced (e.g., 100MB max)
5. File type validation: only `.csv` files accepted
6. Invalid file type returns 400 with message "Only CSV files are allowed"
7. File too large returns 413 with message "File size exceeds 100MB limit"
8. Successful upload returns 202 (Accepted) with `{ingestion_job_id, status: "pending"}`
9. Ingestion job ID stored in PostgreSQL with user_id, filename, status, created_at

```python
"""File upload validation utilities."""
from fastapi import UploadFile, HTTPException, status
from pathlib import Path
import magic  # python-magic library

# Configuration constants
MAX_FILE_SIZE = 100 * 1024 * 1024  # 100MB in bytes
ALLOWED_EXTENSIONS = {'.csv'}
ALLOWED_MIME_TYPES = {'text/csv', 'text/plain', 'application/csv'}

async def validate_csv_upload(file: UploadFile) -> bytes:
    """
    Comprehensive CSV file validation with security checks.

    Security Protections:
    1. File Extension Whitelist - Prevents disguised malicious files
    2. MIME Type Validation - Uses libmagic (not client-provided Content-Type)
    3. File Size Limit - Prevents memory exhaustion
    4. Filename Sanitization - Prevents path traversal attacks

    Args:
        file: Uploaded file from FastAPI

    Returns:
        bytes: Validated file content

    Raises:
        HTTPException: If validation fails
    """
    # 1. Validate filename is not None
    if not file.filename:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Filename is required"
        )

    # 2. Sanitize filename (prevent path traversal)
    sanitized_filename = sanitize_filename(file.filename)
    if sanitized_filename != file.filename:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Invalid filename. Only alphanumeric, hyphens, underscores, and dots allowed."
        )

    # 3. Validate file extension (whitelist)
    file_extension = Path(sanitized_filename).suffix.lower()
    if file_extension not in ALLOWED_EXTENSIONS:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=f"Only CSV files are allowed. Got: {file_extension}"
        )

    # 4. Read file content
    content = await file.read()
    await file.seek(0)  # Reset file pointer for later use

    # 5. Validate file size
    if len(content) > MAX_FILE_SIZE:
        raise HTTPException(
            status_code=status.HTTP_413_REQUEST_ENTITY_TOO_LARGE,
            detail=f"File size exceeds 100MB limit. File size: {len(content) / 1024 / 1024:.2f}MB"
        )

    # 6. Validate MIME type using libmagic (not client-provided Content-Type)
    mime = magic.Magic(mime=True)
    detected_mime = mime.from_buffer(content)

    if detected_mime not in ALLOWED_MIME_TYPES:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=f"Invalid file type. Detected MIME type: {detected_mime}. Expected: CSV file."
        )

    return content

def sanitize_filename(filename: str) -> str:
    """
    Sanitize filename to prevent path traversal and injection attacks.

    Security Rules:
    - Remove path separators (/, \\)
    - Remove null bytes
    - Allow only: alphanumeric, hyphens, underscores, dots
    - Limit length to 255 characters

    Args:
        filename: Original filename from client

    Returns:
        str: Sanitized filename (safe for filesystem)
    """
    import re

    # Remove path components (prevent ../../../etc/passwd attacks)
    filename = Path(filename).name

    # Remove null bytes
    filename = filename.replace('\x00', '')

    # Only allow safe characters: alphanumeric, hyphen, underscore, dot
    safe_pattern = re.compile(r'^[a-zA-Z0-9._-]+$')
    if not safe_pattern.match(filename):
        # Replace unsafe characters with underscores
        filename = re.sub(r'[^a-zA-Z0-9._-]', '_', filename)

    # Limit length
    if len(filename) > 255:
        # Keep extension, truncate name
        name, ext = Path(filename).stem, Path(filename).suffix
        filename = name[:255 - len(ext)] + ext

    return filename
```

### Ingestion Repository

**Create `backend/app/repositories/ingestion_repository.py`:**

```python
"""Ingestion database operations."""
from typing import Optional
from prisma import Prisma
from prisma.models import IngestionJob

class IngestionRepository:
    """Repository for IngestionJob model operations."""

    def __init__(self, db: Prisma):
        self.db = db

    async def create_job(
        self,
        user_id: str,
        file_type: str,
        file_name: str,
        file_size_mb: float
    ) -> IngestionJob:
        """
        Create new ingestion job.

        Args:
            user_id: User who uploaded the file
            file_type: "skills" or "jobs"
            file_name: Original filename
            file_size_mb: File size in megabytes

        Returns:
            Created IngestionJob
        """
        return await self.db.ingestionjob.create(
            data={
                "user_id": user_id,
                "file_type": file_type,
                "file_name": file_name,
                "file_size_mb": file_size_mb,
                "status": "pending"
            }
        )

    async def get_job_by_id(self, job_id: str) -> Optional[IngestionJob]:
        """
        Find ingestion job by ID.

        Args:
            job_id: Job identifier

        Returns:
            IngestionJob if found, None otherwise
        """
        return await self.db.ingestionjob.find_unique(
            where={"id": job_id}
        )

    async def update_job_status(
        self,
        job_id: str,
        status: str
    ) -> IngestionJob:
        """
        Update job status.

        Args:
            job_id: Job identifier
            status: New status value

        Returns:
            Updated IngestionJob
        """
        return await self.db.ingestionjob.update(
            where={"id": job_id},
            data={"status": status}
        )
```

from fastapi import UploadFile
from app.models.ingestion import IngestionJobResponse
from app.repositories.ingestion_repository import IngestionRepository

class IngestionService:
    """Service for CSV ingestion operations."""

    def __init__(self, ingestion_repo: IngestionRepository):
        self.ingestion_repo = ingestion_repo

    async def start_skills_ingestion(
        self,
        file: UploadFile,
        user_id: str
    ) -> IngestionJobResponse:
        """
        Start skills CSV ingestion process.

        Args:
            file: Uploaded CSV file
            user_id: Current authenticated user ID

        Returns:
            IngestionJobResponse with job ID and status
        """
        # Validate file
        content = await validate_csv_upload(file)

        # Calculate file size in MB
        file_size_mb = len(content) / (1024 * 1024)

        # Create ingestion job in PostgreSQL
        job = await self.ingestion_repo.create_job(
            user_id=user_id,
            file_type="skills",
            file_name=file.filename,
            file_size_mb=file_size_mb

```python
"""Ingestion API endpoints."""
from fastapi import APIRouter, UploadFile, File, Depends, status
from app.models.ingestion import IngestionJobResponse
from app.services.ingestion_service import IngestionService
from app.middleware.auth import get_current_user
from app.dependencies import get_ingestion_service

router = APIRouter(prefix="/ingest", tags=["Ingestion"])

@router.post(
    "/skills",
    response_model=IngestionJobResponse,
    status_code=status.HTTP_202_ACCEPTED,
    summary="Upload skills CSV"
)
async def upload_skills_csv(
    file: UploadFile = File(..., description="Skills taxonomy CSV file"),
    user_id: str = Depends(get_current_user),
    ingestion_service: IngestionService = Depends(get_ingestion_service)
) -> IngestionJobResponse:
    """
    Upload and process skills taxonomy CSV.

    File Requirements:
    - Format: CSV (.csv extension)
    - Size: Maximum 100MB
    - Required columns: ID, NAME (others optional)

    Returns:
        202 Accepted: Job created successfully
        400 Bad Request: Invalid file type or format
        401 Unauthorized: Missing or invalid JWT token
        413 Payload Too Large: File exceeds size limit
    """
    return await ingestion_service.start_skills_ingestion(file, user_id)

@router.post(
    "/jobs",
    response_model=IngestionJobResponse,
    status_code=status.HTTP_202_ACCEPTED,
    summary="Upload jobs CSV"
)
async def upload_jobs_csv(
    file: UploadFile = File(..., description="Jobs taxonomy CSV file"),
    user_id: str = Depends(get_current_user),
    ingestion_service: IngestionService = Depends(get_ingestion_service)
) -> IngestionJobResponse:
    """
    Upload and process jobs taxonomy CSV.

    File Requirements:
    - Format: CSV (.csv extension)
    - Size: Maximum 100MB
    - Required columns: Job ID, Job Title, standardized_skills (others optional)

    Returns:
        202 Accepted: Job created successfully
        400 Bad Request: Invalid file type or format
        401 Unauthorized: Missing or invalid JWT token
        413 Payload Too Large: File exceeds size limit
    """
    return await ingestion_service.start_jobs_ingestion(file, user_id)
```

async def validate_file_size(file: UploadFile) -> int:
    file_size = 0
    chunk_size = 1024 * 1024  # 1MB chunks
    await file.seek(0)

    while True:
        chunk = await file.read(chunk_size)
        if not chunk:
            break
        file_size += len(chunk)

        # Early exit on size violation - prevents memory exhaustion
        if file_size > MAX_FILE_SIZE_BYTES:
            raise HTTPException(status_code=413, ...)

    await file.seek(0)  # Reset for subsequent operations
    return file_size
