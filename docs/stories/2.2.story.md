### CSV Parser Utility

**Create `backend/app/utils/csv_parser.py`:**

```python
"""CSV parsing utilities."""
import pandas as pd
import io
from typing import Dict, List

class CSVParseError(Exception):
    """Raised when CSV file cannot be parsed."""
    pass

def parse_csv(content: bytes) -> Dict:
    """
    Parse CSV file content.

    Args:
        content: CSV file content as bytes

    Returns:
        dict: {
            "columns": List[str],  # Column names
            "rows": List[dict],    # All data rows
            "total_count": int     # Total row count
        }

    Raises:
        CSVParseError: If CSV cannot be parsed
    """
    try:
        # Try UTF-8 encoding first
        df = pd.read_csv(io.BytesIO(content), encoding='utf-8')
    except UnicodeDecodeError:
        # Fallback to latin-1 encoding
        try:
            df = pd.read_csv(io.BytesIO(content), encoding='latin-1')
        except Exception as e:
            raise CSVParseError(f"Failed to decode CSV file: {str(e)}")
    except pd.errors.EmptyDataError:
        raise CSVParseError("CSV file is empty")
    except Exception as e:
        raise CSVParseError(f"Failed to parse CSV: {str(e)}")

    # Convert DataFrame to dict format
    return {
        "columns": df.columns.tolist(),
        "rows": df.to_dict(orient='records'),
        "total_count": len(df)
    }
```

### CSV Validation Service

**Create `backend/app/services/csv_validation_service.py`:**

```python
"""CSV content validation service."""
from typing import List, Dict
from app.utils.csv_parser import parse_csv, CSVParseError
from app.exceptions import CSVValidationError

class CSVValidationService:
    """Service for validating CSV content structure."""

    # Required columns for each CSV type (case-insensitive matching)
    SKILLS_REQUIRED_COLUMNS = ["ID", "NAME", "DESCRIPTION", "CATEGORY", "SUBCATEGORY"]
    JOBS_REQUIRED_COLUMNS = ["Job ID", "Job Title", "Company Name", "Location", "standardized_skills"]

    def validate_skills_csv(self, content: bytes) -> Dict:
        """
        Validate skills CSV structure.

        Args:
            content: CSV file content

        Returns:
            dict: Parsed CSV data if valid

        Raises:
            CSVValidationError: If validation fails
        """
        return self._validate_csv(
            content,
            required_columns=self.SKILLS_REQUIRED_COLUMNS,
            csv_type="Skills"
        )

    def validate_jobs_csv(self, content: bytes) -> Dict:
        """
        Validate jobs CSV structure.

        Args:
            content: CSV file content

        Returns:
            dict: Parsed CSV data if valid

        Raises:
            CSVValidationError: If validation fails
        """
        return self._validate_csv(
            content,
            required_columns=self.JOBS_REQUIRED_COLUMNS,
            csv_type="Jobs"
        )

    def _validate_csv(
        self,
        content: bytes,
        required_columns: List[str],
        csv_type: str
    ) -> Dict:
        """
        Generic CSV validation logic.

        Args:
            content: CSV file content
            required_columns: List of required column names
            csv_type: "Skills" or "Jobs" (for error messages)

        Returns:
            dict: Parsed CSV data

        Raises:
            CSVValidationError: If validation fails
        """
        # Parse CSV
        try:
            parsed_data = parse_csv(content)
        except CSVParseError as e:
            raise CSVValidationError(
                detail=f"Failed to parse {csv_type} CSV",
                validation_errors=[str(e)]
            )

        # Check for data rows
        if parsed_data["total_count"] == 0:
            raise CSVValidationError(
                detail="CSV file contains no data rows",
                validation_errors=["CSV has header row but no data"]
            )

        # Check for required columns (case-insensitive)
        csv_columns_lower = [col.lower().strip() for col in parsed_data["columns"]]
        required_columns_lower = [col.lower() for col in required_columns]

        missing_columns = []
        for required_col in required_columns_lower:
            if required_col not in csv_columns_lower:
                # Find original casing for error message
                original_name = next(
                    (col for col in required_columns if col.lower() == required_col),
                    required_col
                )
                missing_columns.append(original_name)

        if missing_columns:
            raise CSVValidationError(
                detail=f"Missing required columns in {csv_type} CSV",
                validation_errors=[
                    f"Required columns: {', '.join(required_columns)}",
                    f"Missing columns: {', '.join(missing_columns)}",
                    f"Found columns: {', '.join(parsed_data['columns'])}"
                ]
            )

        return parsed_data
```

### Update Ingestion Service

**Update `backend/app/services/ingestion_service.py`:**

```python
from app.services.csv_validation_service import CSVValidationService

class IngestionService:
    """Service for CSV ingestion operations."""

    def __init__(
        self,
        ingestion_repo: IngestionRepository,
        validation_service: CSVValidationService
    ):
        self.ingestion_repo = ingestion_repo
        self.validation_service = validation_service

    async def start_skills_ingestion(
        self,
        file: UploadFile,
        user_id: str
    ) -> IngestionJobResponse:
        """Start skills CSV ingestion with validation."""
        # Validate file format
        content = await validate_csv_upload(file)

        # Validate CSV structure
        parsed_data = self.validation_service.validate_skills_csv(content)

        # Create ingestion job
        file_size_mb = len(content) / (1024 * 1024)
        job = await self.ingestion_repo.create_job(
            user_id=user_id,
            file_type="skills",
            file_name=file.filename,
            file_size_mb=file_size_mb
        )

        # Update job with total records count
        await self.ingestion_repo.update_job(
            job_id=job.id,
            total_records=parsed_data["total_count"]
        )

        return IngestionJobResponse(
            ingestion_job_id=job.id,
            status="pending",
            message=f"Skills CSV validated successfully. {parsed_data['total_count']} records ready for processing."
        )
```

```python
def sanitize_csv_value(value: Any) -> Any:
    """Sanitize CSV cell value to prevent formula injection attacks."""
    if not isinstance(value, str):
        return value

    dangerous_chars = ('=', '+', '-', '@', '|', '%')
    if value.strip().startswith(dangerous_chars):
        sanitized = "'" + value
        logger.debug(f"Sanitized potential formula injection: {value[:20]}...")
        return sanitized

    return value
```

   - **Example**: A cell containing `=1+1` or `=cmd|'/c calc'!A1` could execute commands
   - **Recommendation**: Add sanitization to prefix dangerous characters with single quote (')

2. **Broad Exception Handling (Medium Risk)**
   - **Location**: csv_parser.py:36
   - **Issue**: `except Exception as e:` catches all exceptions, could mask security issues
   - **Recommendation**: Use specific exception types (pandas.errors.ParserError, etc.)

3. **No Logging (Medium Risk)**
   - **Issue**: Critical validation operations have no audit logging
   - **Recommendation**: Add structured logging for validation failures, suspicious patterns

### Performance Considerations

**CONCERNS IDENTIFIED:**

1. **Memory Loading of Large Files**
   - **Issue**: `df.to_dict(orient='records')` loads entire dataset into memory
   - **Impact**: Files approaching 100MB limit could cause OOM errors
   - **Current**: Works for typical files, but not scalable
   - **Recommendation**: Consider chunked reading or streaming for Story 2.4 (Batch Processing)

2. **No Streaming Support**
   - **Issue**: All CSV data loaded at once
   - **Impact**: Memory usage scales linearly with file size
   - **Recommendation**: Investigate pandas `chunksize` parameter for large file handling
