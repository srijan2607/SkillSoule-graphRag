# Story 2.3: CSV Preview & Confirmation

## Status

**Ready for Review**

## Story

**As a** user,
**I want** to preview the first few rows of my CSV before confirming ingestion,
**so that** I can verify the data looks correct

## Acceptance Criteria

1. After validation passes, extract first 5-10 rows from CSV
2. Return preview data to frontend: `{columns: [...], preview_rows: [...], total_rows: N}`
3. Preview includes column names and sample values
4. Row count calculated and returned (e.g., "40,523 jobs found")
5. Frontend displays preview in table format
6. Frontend provides "Confirm Upload" and "Cancel" buttons
7. Cancel button discards uploaded file, no data stored
8. Confirm button triggers actual ingestion process

```python
"""CSV parsing utilities with preview support."""
import pandas as pd
import io
from typing import Dict, List

class CSVParseError(Exception):
    """Raised when CSV file cannot be parsed."""
    pass

def parse_csv(content: bytes) -> Dict:
    """
    Parse complete CSV file content.
    (Already implemented in Story 2.2)
    """
    try:
        df = pd.read_csv(io.BytesIO(content), encoding='utf-8')
    except UnicodeDecodeError:
        try:
            df = pd.read_csv(io.BytesIO(content), encoding='latin-1')
        except Exception as e:
            raise CSVParseError(f"Failed to decode CSV file: {str(e)}")
    except pd.errors.EmptyDataError:
        raise CSVParseError("CSV file is empty")
    except Exception as e:
        raise CSVParseError(f"Failed to parse CSV: {str(e)}")

    return {
        "columns": df.columns.tolist(),
        "rows": df.to_dict(orient='records'),
        "total_count": len(df)
    }

def parse_csv_preview(content: bytes, preview_rows: int = 10) -> Dict:
    """
    Parse CSV file and return preview with first N rows.

    Args:
        content: CSV file content as bytes
        preview_rows: Number of rows to include in preview (default: 10)

    Returns:
        dict: {
            "columns": List[str],        # Column names
            "preview_rows": List[dict],  # First N data rows
            "total_rows": int,           # Total row count
            "has_more": bool             # True if total_rows > preview_rows
        }

    Raises:
        CSVParseError: If CSV cannot be parsed
    """
    try:
        # Try UTF-8 encoding first
        df = pd.read_csv(io.BytesIO(content), encoding='utf-8')
    except UnicodeDecodeError:
        # Fallback to latin-1 encoding
        try:
            df = pd.read_csv(io.BytesIO(content), encoding='latin-1')
        except Exception as e:
            raise CSVParseError(f"Failed to decode CSV file: {str(e)}")
    except pd.errors.EmptyDataError:
        raise CSVParseError("CSV file is empty")
    except Exception as e:
        raise CSVParseError(f"Failed to parse CSV: {str(e)}")

    total_rows = len(df)

    # Get first N rows for preview
    preview_df = df.head(preview_rows)

    return {
        "columns": df.columns.tolist(),
        "preview_rows": preview_df.to_dict(orient='records'),
        "total_rows": total_rows,
        "has_more": total_rows > preview_rows
    }
```

```python
"""CSV content validation service with preview support."""
from typing import Dict
from app.utils.csv_parser import parse_csv_preview, CSVParseError
from app.exceptions import CSVValidationError

class CSVValidationService:
    """Service for validating CSV content structure with preview."""

    # Required columns (from Epic 2 PRD - Story 2.2)
    # ⚠️ CRITICAL: These MUST match Story 2.2's actual implementation
    SKILLS_REQUIRED_COLUMNS = ["ID", "NAME", "DESCRIPTION", "CATEGORY", "SUBCATEGORY"]
    JOBS_REQUIRED_COLUMNS = ["Job ID", "Job Title", "Company Name", "Location", "standardized_skills"]

    def validate_and_preview_skills(
        self,
        content: bytes,
        preview_rows: int = 10
    ) -> Dict:
        """
        Validate skills CSV and return preview data.

        Args:
            content: CSV file content
            preview_rows: Number of rows to preview

        Returns:
            dict: {
                "is_valid": bool,
                "columns": List[str],
                "preview_rows": List[dict],
                "total_rows": int,
                "has_more": bool,
                "validation_errors": List[str]  # Empty if valid
            }

        Raises:
            CSVValidationError: If validation fails
        """
        return self._validate_and_preview(
            content,
            required_columns=self.SKILLS_REQUIRED_COLUMNS,
            csv_type="Skills",
            preview_rows=preview_rows
        )

    def validate_and_preview_jobs(
        self,
        content: bytes,
        preview_rows: int = 10
    ) -> Dict:
        """
        Validate jobs CSV and return preview data.

        Args:
            content: CSV file content
            preview_rows: Number of rows to preview

        Returns:
            dict: Preview data with validation status

        Raises:
            CSVValidationError: If validation fails
        """
        return self._validate_and_preview(
            content,
            required_columns=self.JOBS_REQUIRED_COLUMNS,
            csv_type="Jobs",
            preview_rows=preview_rows
        )

    def _validate_and_preview(
        self,
        content: bytes,
        required_columns: List[str],
        csv_type: str,
        preview_rows: int
    ) -> Dict:
        """
        Generic CSV validation and preview logic.

        Args:
            content: CSV file content
            required_columns: List of required column names
            csv_type: "Skills" or "Jobs" (for error messages)
            preview_rows: Number of rows to preview

        Returns:
            dict: Preview data with validation status

        Raises:
            CSVValidationError: If validation fails
        """
        # Parse CSV with preview
        try:
            preview_data = parse_csv_preview(content, preview_rows)
        except CSVParseError as e:
            raise CSVValidationError(
                detail=f"Failed to parse {csv_type} CSV",
                validation_errors=[str(e)]
            )

        validation_errors = []

        # Check for data rows
        if preview_data["total_rows"] == 0:
            raise CSVValidationError(
                detail="CSV file contains no data rows",
                validation_errors=["CSV has header row but no data"]
            )

        # Check for required columns (case-insensitive)
        csv_columns_lower = [col.lower().strip() for col in preview_data["columns"]]
        required_columns_lower = [col.lower() for col in required_columns]

        missing_columns = []
        for required_col in required_columns_lower:
            if required_col not in csv_columns_lower:
                # Find original casing for error message
                original_name = next(
                    (col for col in required_columns if col.lower() == required_col),
                    required_col
                )
                missing_columns.append(original_name)

        if missing_columns:
            raise CSVValidationError(
                detail=f"Missing required columns in {csv_type} CSV",
                validation_errors=[
                    f"Required columns: {', '.join(required_columns)}",
                    f"Missing columns: {', '.join(missing_columns)}",
                    f"Found columns: {', '.join(preview_data['columns'])}"
                ]
            )

        # Validation successful
        return {
            "is_valid": True,
            "columns": preview_data["columns"],
            "preview_rows": preview_data["preview_rows"],
            "total_rows": preview_data["total_rows"],
            "has_more": preview_data["has_more"],
            "validation_errors": []
        }
```

### Temporary File Storage

**Create `backend/app/utils/file_cache.py`:**

```python
"""Temporary file storage for CSV previews."""
import hashlib
from typing import Optional
from datetime import datetime, timedelta

# Simple in-memory cache for MVP (replace with Redis for production)
_file_cache: Dict[str, Dict] = {}

def store_temp_file(
    content: bytes,
    file_type: str,
    filename: str,
    ttl_hours: int = 1
) -> str:
    """
    Store file content temporarily with TTL.

    Args:
        content: File content as bytes
        file_type: "skills" or "jobs"
        filename: Original filename (for ingestion job creation)
        ttl_hours: Time to live in hours (default: 1)

    Returns:
        File hash (SHA-256) as identifier
    """
    # Generate SHA-256 hash
    file_hash = hashlib.sha256(content).hexdigest()

    # Calculate file size
    file_size_mb = len(content) / (1024 * 1024)

    # Store in cache with expiry and metadata
    _file_cache[file_hash] = {
        "content": content,
        "file_type": file_type,
        "filename": filename,  # ADDED: Preserve original filename
        "file_size_mb": file_size_mb,  # ADDED: For ingestion job
        "expires_at": datetime.utcnow() + timedelta(hours=ttl_hours)
    }

    return file_hash

def retrieve_temp_file(file_hash: str) -> Optional[bytes]:
    """
    Retrieve temporary file content by hash.

    Args:
        file_hash: SHA-256 hash identifier

    Returns:
        File content as bytes, or None if not found/expired
    """
    if file_hash not in _file_cache:
        return None

    cached_file = _file_cache[file_hash]

    # Check if expired
    if datetime.utcnow() > cached_file["expires_at"]:
        del _file_cache[file_hash]
        return None

    return cached_file["content"]

def delete_temp_file(file_hash: str) -> bool:
    """
    Delete temporary file from cache.

    Args:
        file_hash: SHA-256 hash identifier

    Returns:
        True if deleted, False if not found
    """
    if file_hash in _file_cache:
        del _file_cache[file_hash]
        return True
    return False

def cleanup_expired_files() -> int:
    """
    Remove all expired files from cache.

    Returns:
        Number of files cleaned up
    """
    expired_hashes = [
        file_hash
        for file_hash, data in _file_cache.items()
        if datetime.utcnow() > data["expires_at"]
    ]

    for file_hash in expired_hashes:
        del _file_cache[file_hash]

    return len(expired_hashes)
```

```python
from fastapi import APIRouter, UploadFile, File, Depends, HTTPException, status
from app.models.ingestion import (
    CSVPreviewResponse,
    ConfirmUploadRequest,
    IngestionJobResponse
)
from app.services.csv_validation_service import CSVValidationService
from app.services.ingestion_service import IngestionService
from app.middleware.auth import get_current_user
from app.utils.file_validation import validate_csv_upload
from app.utils.file_cache import store_temp_file, retrieve_temp_file

router = APIRouter(prefix="/api/ingest", tags=["ingestion"])

@router.post("/skills", response_model=CSVPreviewResponse, status_code=200)
async def upload_skills_csv(
    file: UploadFile = File(...),
    user_id: str = Depends(get_current_user),
    validation_service: CSVValidationService = Depends()
) -> CSVPreviewResponse:
    """
    Upload skills CSV and return preview (does not start ingestion).

    Returns preview data with first 10 rows. User must confirm via
    POST /ingest/confirm to start actual ingestion.
    """
    # Validate file format
    content = await validate_csv_upload(file)

    # Validate CSV structure and get preview
    preview_data = validation_service.validate_and_preview_skills(
        content,
        preview_rows=10
    )

    # Store file temporarily for confirmation (with metadata)
    file_hash = store_temp_file(
        content,
        file_type="skills",
        filename=file.filename,
        ttl_hours=1
    )

    return CSVPreviewResponse(
        **preview_data,
        file_hash=file_hash,
        file_type="skills"
    )

@router.post("/jobs", response_model=CSVPreviewResponse, status_code=200)
async def upload_jobs_csv(
    file: UploadFile = File(...),
    user_id: str = Depends(get_current_user),
    validation_service: CSVValidationService = Depends()
) -> CSVPreviewResponse:
    """
    Upload jobs CSV and return preview (does not start ingestion).

    Returns preview data with first 10 rows. User must confirm via
    POST /ingest/confirm to start actual ingestion.
    """
    # Validate file format
    content = await validate_csv_upload(file)

    # Validate CSV structure and get preview
    preview_data = validation_service.validate_and_preview_jobs(
        content,
        preview_rows=10
    )

    # Store file temporarily for confirmation
    file_hash = store_temp_file(content, file_type="jobs", ttl_hours=1)

    return CSVPreviewResponse(
        **preview_data,
        file_hash=file_hash,
        file_type="jobs"
    )

@router.post("/confirm", response_model=IngestionJobResponse, status_code=202)
async def confirm_upload(
    confirmation: ConfirmUploadRequest,
    user_id: str = Depends(get_current_user),
    ingestion_service: IngestionService = Depends()
) -> IngestionJobResponse:
    """
    Confirm CSV upload and start ingestion process.

    Retrieves temporarily stored file and creates IngestionJob.
    """
    # Retrieve file from temporary storage
    content = retrieve_temp_file(confirmation.file_hash)

    if not content:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="File not found or expired. Please upload again."
        )

    # Start ingestion based on file type
    if confirmation.file_type == "skills":
        response = await ingestion_service.start_skills_ingestion(content, user_id)
    elif confirmation.file_type == "jobs":
        response = await ingestion_service.start_jobs_ingestion(content, user_id)
    else:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=f"Invalid file_type: {confirmation.file_type}"
        )

    # Clean up temporary file
    delete_temp_file(confirmation.file_hash)

    return response
```

### Production Warnings

**⚠️ IN-MEMORY CACHE LIMITATIONS**

The `_file_cache` dict is a **simple MVP solution** with limitations:

**NOT Production-Ready:**
- ❌ Lost on server restart (no persistence)
- ❌ Memory pressure with large files (100MB × N users = OOM risk)
- ❌ No distributed support (won't work with multiple server instances)
- ❌ No thread-safety for concurrent access

**Maximum Concurrent Uploads:**
- Recommended: 10-20 concurrent users (1-2GB memory for 100MB files)
- Monitor memory usage closely
- Implement rate limiting

**Production Replacement (Future Story):**

```python
# Use Redis with automatic TTL
import redis

redis_client = redis.Redis(host='localhost', port=6379, db=0)

def store_temp_file(content: bytes, file_type: str, filename: str, ttl_hours: int = 1):
    """Store file in Redis with automatic expiry."""
    file_hash = hashlib.sha256(content).hexdigest()

    # Store with metadata as JSON
    metadata = {
        "content": content.decode('latin-1'),  # or base64 encode
        "file_type": file_type,
        "filename": filename,
        "file_size_mb": len(content) / (1024 * 1024)
    }

    # Set with TTL (automatic cleanup)
    redis_client.setex(
        f"upload:{file_hash}",
        ttl_hours * 3600,
        json.dumps(metadata)
    )

    return file_hash
```

```python
from apscheduler.schedulers.background import BackgroundScheduler
from app.utils.file_cache import cleanup_expired_files
import logging

logger = logging.getLogger(__name__)

# Create scheduler
scheduler = BackgroundScheduler()

def cleanup_task():
    """Scheduled cleanup task."""
    try:
        cleaned = cleanup_expired_files()
        if cleaned > 0:
            logger.info(f"Cleaned up {cleaned} expired temporary files")
    except Exception as e:
        logger.error(f"Cleanup task failed: {str(e)}", exc_info=True)

# Schedule cleanup every 15 minutes

    """Start background scheduler."""
    scheduler.start()
    logger.info("File cleanup scheduler started (runs every 15 minutes)")

    """Stop background scheduler."""
```
