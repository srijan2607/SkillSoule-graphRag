### Example Metrics Implementation
```python
# app/utils/metrics.py
import time
from typing import Dict
from dataclasses import dataclass, field

@dataclass
class QueryMetrics:
    """Collect and track query performance metrics."""

    query_id: str
    user_id: str
    timers: Dict[str, float] = field(default_factory=dict)

    def start_timer(self, node_name: str):
        """Start timer for a node."""
        self.timers[f"{node_name}_start"] = time.time()

    def end_timer(self, node_name: str):
        """End timer for a node and calculate duration."""
        start_key = f"{node_name}_start"
        if start_key in self.timers:
            duration_ms = (time.time() - self.timers[start_key]) * 1000
            self.timers[f"{node_name}_duration_ms"] = duration_ms
            del self.timers[start_key]
            return duration_ms
        return 0

    def get_summary(self) -> Dict[str, float]:
        """Get metrics summary."""
        return {k: v for k, v in self.timers.items() if k.endswith("_duration_ms")}

    def check_performance_targets(self) -> Dict[str, bool]:
        """Check if performance targets met."""
        summary = self.get_summary()
        return {
            "vector_search_ok": summary.get("vector_search_duration_ms", 0) < 1000,
            "graph_traversal_ok": summary.get("graph_traversal_duration_ms", 0) < 2000,
            "total_ok": summary.get("total_duration_ms", 0) < 5000
        }
```

### Node Instrumentation Pattern
```python
# In each LangGraph node
from app.utils.metrics import QueryMetrics

async def vector_search_node(state: GraphRAGState) -> dict:
    metrics = state.metadata.get("metrics")

    if metrics:
        metrics.start_timer("vector_search")

    # Node logic here
    results = await neo4j_repo.vector_search(...)

    if metrics:
        duration = metrics.end_timer("vector_search")
        logger.info(f"Vector search completed in {duration:.2f}ms")

    return {"vector_results": results}
```

### Structured Logging Example
```python
import json
import logging

logger = logging.getLogger(__name__)

# After query execution
metrics_summary = metrics.get_summary()
logger.info(
    json.dumps({
        "event": "query_completed",
        "query_id": query_id,
        "user_id": user_id,
        "metrics": metrics_summary,
        "performance_targets": metrics.check_performance_targets()
    })
)

# Slow query alert
if metrics_summary.get("total_duration_ms", 0) > 5000:
    logger.warning(
        json.dumps({
            "event": "slow_query_alert",
            "query_id": query_id,
            "query_text": user_query,
            "total_time_ms": metrics_summary["total_duration_ms"],
            "breakdown": metrics_summary
        })
    )
```

### Critical Implementation Notes
1. **Non-intrusive**: Metrics collection shouldn't slow down queries
2. **Optional storage**: PostgreSQL storage is optional for MVP
3. **Structured logging**: Use JSON format for easy parsing
4. **Performance targets**: Informational, not hard limits (don't fail queries)
5. **Alert threshold**: 5 seconds is soft limit, log warning only
6. **Async logging**: PostgreSQL writes should be async/non-blocking
