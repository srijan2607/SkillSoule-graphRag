### Component Structure
```typescript
// src/components/Chat/SourceCitations.tsx
import { useState } from "react";
import { SourceNode } from "@/types/chat";

interface SourceCitationsProps {
  sources: SourceNode[];
}

export default function SourceCitations({ sources }: SourceCitationsProps) {
  const [isExpanded, setIsExpanded] = useState(false);

  const grouped = groupSourcesByType(sources);
  const summary = formatSourceSummary(grouped);

  return (
    <div className="source-citations mt-2 border-t border-gray-200 pt-2">
      <button
        onClick={() => setIsExpanded(!isExpanded)}
        className="flex items-center text-sm text-gray-600 hover:text-gray-900"
        aria-expanded={isExpanded}
      >
        <ChevronIcon className={isExpanded ? "rotate-180" : ""} />
        <span className="ml-2">{summary}</span>
      </button>

      {isExpanded && (
        <div className="mt-2 space-y-2">
          {grouped.Skills.length > 0 && (
            <SourceGroup title="Skills" sources={grouped.Skills} />
          )}
          {grouped.Jobs.length > 0 && (
            <SourceGroup title="Jobs" sources={grouped.Jobs} />
          )}
          {grouped.Companies.length > 0 && (
            <SourceGroup title="Companies" sources={grouped.Companies} />
          )}
        </div>
      )}
    </div>
  );
}
```

### Utility Functions
```typescript
// src/utils/sources.ts
interface GroupedSources {
  Skills: SourceNode[];
  Jobs: SourceNode[];
  Companies: SourceNode[];
}

export function groupSourcesByType(sources: SourceNode[]): GroupedSources {
  return sources.reduce(
    (acc, source) => {
      const type = source.node_type;
      if (type === "Skill") acc.Skills.push(source);
      else if (type === "Job") acc.Jobs.push(source);
      else if (type === "Company") acc.Companies.push(source);
      return acc;
    },
    { Skills: [], Jobs: [], Companies: [] }
  );
}

export function formatSourceSummary(grouped: GroupedSources): string {
  const parts = [];
  if (grouped.Jobs.length > 0) {
    parts.push(`${grouped.Jobs.length} job${grouped.Jobs.length > 1 ? "s" : ""}`);
  }
  if (grouped.Skills.length > 0) {
    parts.push(`${grouped.Skills.length} skill${grouped.Skills.length > 1 ? "s" : ""}`);
  }
  if (grouped.Companies.length > 0) {
    parts.push(`${grouped.Companies.length} ${grouped.Companies.length > 1 ? "companies" : "company"}`);
  }

  return parts.length > 0 ? `Based on ${parts.join(", ")}` : "No sources";
}
```

### SourceGroup Component
```typescript
// src/components/Chat/SourceGroup.tsx
interface SourceGroupProps {
  title: string;
  sources: SourceNode[];
}

export default function SourceGroup({ title, sources }: SourceGroupProps) {
  return (
    <div className="source-group">
      <h4 className="text-xs font-semibold text-gray-700 mb-1">{title}</h4>
      <div className="space-y-1">
        {sources.map((source, idx) => (
          <div key={idx} className="text-xs text-gray-600 pl-2">
            • {source.properties.name || source.properties.job_title || source.node_id}
          </div>
        ))}
      </div>
    </div>
  );
}
```
