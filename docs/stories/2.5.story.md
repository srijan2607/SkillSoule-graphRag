# Story 2.5: Real-Time Progress Tracking

## Status

**Done**

## Story

**As a** user,
**I want** to see real-time progress while my CSV is being ingested,
**so that** I know the system is working and how long it will take

## Acceptance Criteria

1. `GET /ingest/status/{job_id}` endpoint created (protected)
2. Endpoint returns JSON: `{status, records_processed, records_failed, total_records, estimated_time_remaining}`
3. Frontend polls `/ingest/status/{job_id}` every 2 seconds during ingestion
4. Progress bar displays percentage: `(records_processed / total_records) * 100`
5. Display stats: "Processing 15,234 / 40,523 records (5 failed)"
6. Estimated time remaining calculated based on average processing speed
7. Status values: "pending", "processing", "completed", "completed_with_errors", "failed"
8. When status = "completed", polling stops and success message shown

### Status Endpoint

**Update `backend/app/api/ingest.py`:**

```python
from fastapi import APIRouter, Depends, HTTPException, status
from app.models.ingestion import IngestionStatusResponse
from app.services.ingestion_service import IngestionService
from app.middleware.auth import get_current_user

router = APIRouter(prefix="/api/ingest", tags=["ingestion"])

@router.get("/status/{job_id}", response_model=IngestionStatusResponse)
async def get_ingestion_status(
    job_id: str,
    user_id: str = Depends(get_current_user),
    ingestion_service: IngestionService = Depends()
) -> IngestionStatusResponse:
    """
    Poll ingestion job status and progress.

    Frontend should poll this endpoint every 2 seconds while status
    is "pending" or "processing".

    Polling stops when status is "completed", "completed_with_errors", or "failed".

    Authorization:
    - User must own the job (user_id must match job.user_id)
    """
    return await ingestion_service.get_job_status(job_id, user_id)
```

### Status Calculation Service

**Update `backend/app/services/ingestion_service.py`:**

```python
from datetime import datetime
from app.models.ingestion import IngestionStatusResponse
from app.repositories.ingestion_repository import IngestionRepository
from fastapi import HTTPException, status

class IngestionService:
    # ... existing methods ...

    async def get_job_status(
        self,
        job_id: str,
        user_id: str
    ) -> IngestionStatusResponse:
        """
        Get ingestion job status with progress and ETA.

        Args:
            job_id: Ingestion job ID
            user_id: Current user ID (for authorization)

        Returns:
            IngestionStatusResponse with current status and progress

        Raises:
            404: Job not found
            403: User doesn't own the job
        """
        # Fetch job from database
        job = await self.ingestion_repo.get_job(job_id)

        if not job:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail=f"Ingestion job {job_id} not found"
            )

        # Authorization check
        if job.user_id != user_id:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="You do not have permission to access this job"
            )

        # Calculate progress percentage
        progress_percentage = 0.0
        if job.total_records > 0:
            progress_percentage = (
                job.processed_records / job.total_records
            ) * 100

        # Calculate estimated time remaining
        estimated_time_remaining = None
        if job.status == "processing" and job.processing_speed:
            remaining_records = job.total_records - job.processed_records
            estimated_time_remaining = int(
                remaining_records / job.processing_speed
            )

        # Calculate total batches
        total_batches = None
        if job.total_records > 0:
            batch_size = settings.BATCH_SIZE
            total_batches = (
                job.total_records + batch_size - 1
            ) // batch_size

        return IngestionStatusResponse(
            job_id=job.id,
            status=job.status,
            total_records=job.total_records,
            processed_records=job.processed_records,
            failed_records=job.failed_records,
            current_batch=job.batch_number,
            total_batches=total_batches,
            progress_percentage=round(progress_percentage, 2),
            estimated_time_remaining=estimated_time_remaining,
            processing_speed=job.processing_speed,
            started_at=job.started_at,
            completed_at=job.completed_at,
            error_message=job.error_log
        )
```

### Processing Speed Calculation

**Update `backend/app/services/batch_processor.py`:**

```python
import time
from datetime import datetime, timedelta

class BatchProcessor:
    # ... existing methods ...

    async def process_skills_batch(
        self,
        batch: List[Dict[str, Any]],
        job_id: str,
        batch_number: int,
        start_row: int
    ) -> Dict[str, Any]:
        """
        Process a batch of skill records with speed tracking.

        Returns batch statistics including processing time.
        """
        batch_start_time = time.time()

        # ... existing batch processing logic ...

        batch_end_time = time.time()
        batch_duration = batch_end_time - batch_start_time
        batch_speed = len(batch) / batch_duration  # records per second

        # Update job with speed metrics
        await self._update_processing_speed(
            job_id=job_id,
            batch_speed=batch_speed,
            processed_count=processed
        )

        return {
            "processed": processed,
            "failed": failed,
            "errors": errors,
            "duration": batch_duration,
            "speed": batch_speed
        }

    async def _update_processing_speed(
        self,
        job_id: str,
        batch_speed: float,
        processed_count: int
    ) -> None:
        """
        Update job processing speed with exponential moving average.

        Args:
            job_id: Ingestion job ID
            batch_speed: Current batch speed (records/sec)
            processed_count: Total records processed so far
        """
        # Fetch current job
        job = await self.ingestion_repo.get_job(job_id)

        # Calculate exponential moving average (alpha = 0.3)
        if job.processing_speed is None:
            # First batch - use actual speed
            new_speed = batch_speed
        else:
            # EMA: new_speed = alpha * batch_speed + (1 - alpha) * old_speed
            alpha = 0.3
            new_speed = alpha * batch_speed + (1 - alpha) * job.processing_speed

        # Calculate ETA
        remaining_records = job.total_records - processed_count
        eta_seconds = remaining_records / new_speed
        estimated_completion_time = datetime.utcnow() + timedelta(seconds=eta_seconds)

        # Update job
        await self.ingestion_repo.update_job(
            job_id=job_id,
            processing_speed=new_speed,
            estimated_completion_time=estimated_completion_time
        )
```
