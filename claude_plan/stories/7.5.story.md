### NetworkInsights Model Schema

```python
# backend/app/models/query.py

from typing import List, Optional, Dict, Any
from pydantic import BaseModel


class SkillPathResult(BaseModel):
    """Result of skill path calculation."""
    from_skill: str
    to_skill: str
    path: List[str]  # ["Python", "Data Analysis", "Machine Learning"]
    total_cost: float  # Sum of edge costs
    closeness: float  # 1 / (1 + total_cost)


class SimilarJobResult(BaseModel):
    """Result of job similarity lookup."""
    job_id: str
    job_title: str
    company: Optional[str] = None
    jaccard_score: float
    shared_skills: List[str]


class TopSkillResult(BaseModel):
    """Skill ranked by importance metrics."""
    skill_name: str
    canonical_name: str
    centrality: float
    demand_count: int


class NetworkInsights(BaseModel):
    """Network-based insights from graph analysis."""
    skill_paths: List[SkillPathResult] = []
    similar_jobs: List[SimilarJobResult] = []
    top_skills: List[TopSkillResult] = []
    transition_feasibility: Optional[float] = None
    graph_stats: Optional[Dict[str, Any]] = None


class QueryResponse(BaseModel):
    """Response from query endpoint."""
    query: str
    response: str
    sources: List[Dict[str, Any]]
    processing_time_ms: float
    metadata: Dict[str, Any]
    network_insights: Optional[NetworkInsights] = None  # NEW - nullable
```

### Context Section Format

```python
# Network Insights section in context_construction.py

def _build_network_insights_section(enrichment: Dict) -> str:
    """Build Network Insights section for LLM context."""
    if not enrichment:
        return ""

    sections = []
    sections.append("## Network Insights")
    sections.append("")

    # Skill Paths
    if enrichment.get("skill_paths"):
        sections.append("**Skill Bridge Paths:**")
        for path in enrichment["skill_paths"][:3]:
            path_str = " → ".join(path["path"])
            sections.append(f"- {path_str} (closeness: {path['closeness']:.2f})")
        sections.append("")

    # Top Skills
    if enrichment.get("top_skills_by_centrality"):
        sections.append("**Top Skills by Importance:**")
        for i, skill in enumerate(enrichment["top_skills_by_centrality"][:5], 1):
            sections.append(f"{i}. {skill['name']} (centrality: {skill['centrality']:.2f})")
        sections.append("")

    # Similar Jobs
    if enrichment.get("similar_jobs"):
        sections.append("**Similar Job Opportunities:**")
        for job in enrichment["similar_jobs"][:5]:
            match_pct = int(job["jaccard_score"] * 100)
            sections.append(f"- {job['job_title']} ({match_pct}% skill match)")
        sections.append("")

    return "\n".join(sections)
```

### Hybrid Scoring Formula

```
hybrid_score = 0.50 * vector_score + 0.30 * centrality + 0.20 * demand_normalized

Where:
- vector_score: Cosine similarity from vector search (0.0 - 1.0)
- centrality: Eigenvector centrality from graph (0.0 - 1.0)
- demand_normalized: demand_count / max_demand_count (0.0 - 1.0)
```

### Backward Compatibility

- `network_insights` field is Optional (nullable)
- Existing clients that don't expect this field will ignore it
- If no network enrichment occurred, field is `null` or omitted
